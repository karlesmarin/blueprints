(*  Title:      Memory_Time_Bound.thy    Author:     Carles Marín    Copyright (c) 2026 Carles Marín. All rights reserved.    License:    BSD     Corollary cor:time of paper A (memory floors).  The write infidelity of a pulse is r = (1-p)/6 + (1-c)/3, where    p is the transferred population and c the coherence.  If every pulse of duration T has p <= Bp T and c <= Bc T,    then r >= r_time = 1/2 - S/6 with S = sup_{T>=0} (Bp T + 2 Bc T).  If the sup is attained at T0 where the    resonant pulse has pres T0 = Bp T0 and cres T0 = Bc T0 (or approached along T_n with the gap tending to 0),    the resonant pulse reaches r_time (in the limit). Pointwise bound. *)lemma time_pointwise:  fixes p c :: real and Bp Bc :: "real \<Rightarrow> real"  assumes "p \<le> Bp T" and "c \<le> Bc T"  shows "1 / 2 - (Bp T + 2 * Bc T) / 6 \<le> (1 - p) / 6 + (1 - c) / 3"  using write_channel_monotone[where p = p and c = c and P = "Bp T" and C = "Bc T"] assms  by argo (* 2.

(1 - t) * inf_affine I \<alpha> \<beta> r + t * inf_affine I \<alpha> \<beta> s                \<le> \<alpha> i + \<beta> i * ((1 - t) *\<^sub>R r + t *\<^sub>R s)"  proof    fix i assume i: "i \<in> I"    have a: "inf_affine I \<alpha> \<beta> r \<le> \<alpha> i + \<beta> i * r"      using inf_affine_le[where r = r and i = i] assms(2) rs(1) i by simp    have b: "inf_affine I \<alpha> \<beta> s \<le> \<alpha> i + \<beta> i * s"      using inf_affine_le[where r = s and i = i] assms(2) rs(2) i by simp    have "(1 - t) * inf_affine I \<alpha> \<beta> r + t * inf_affine I \<alpha> \<beta> s          \<le> (1 - t) * (\<alpha> i + \<beta> i * r) + t * (\<alpha> i + \<beta> i * s)"      using a b t by (intro add_mono mult_left_mono) simp_all    also have "\<dots> = \<alpha> i + \<beta> i * ((1 - t) *\<^sub>R r + t *\<^sub>R s)" by (simp add: real_scaleR_def algebra_simps)    finally show "(1 - t) * inf_affine I \<alpha> \<beta> r + t * inf_affine I \<alpha> \<beta> s                  \<le> \<alpha> i + \<beta> i * ((1 - t) *\<^sub>R r + t *\<^sub>R s)" .  qed  then show "(1 - t) * inf_affine I \<alpha> \<beta> r + t * inf_affine I \<alpha> \<beta> s             \<le> inf_affine I \<alpha> \<beta> ((1 - t) *\<^sub>R r + t *\<^sub>R s)"    unfolding inf_affine_def using assms(1) by (intro cInf_greatest) autoqed (simp add: convex_real_interval) end

License:    BSD*) (* encaje: orthogonality of the rational coefficient vectors of a diagonal form with C4-free support *)section \<open>Rational representations of a diagonal form with $C_4$-free support\<close> theory BSR_Rational  imports Coefficient_Vectors "HOL-Library.Function_Algebras"begin text \<open>A representation of $f=\sum_{(i,j)\in E}c_{ij}x_i^2y_j^2$ as a sum of $r$ squares of bilinear  forms $s_k=\sum_{ij}v_{ij}(k)\,x_iy_j$ has coefficient vectors $v_{ij}\in\mathbb{Q}^r$. Comparing  the coefficients of the monomial $x_ix_ky_jy_l$ in $\sum_k s_k^2=f$ gives the polarized identity  $v_{ij}\cdot v_{kl}+v_{il}\cdot v_{kj}=2\,[i=k\wedge j=l]\,c_{ij}$ (Zhang and Zhu  @{cite ZhangZhu}, Prop.~2.1, for general coefficients, with $c=0$ off $E$). This identity is  taken as the definition \<open>polarized\<close> below; the comparison of coefficients that leads to it  is not formalized.   The squares are indexed by a finite type \<open>'r\<close>, the vectors are functions \<open>'r \<Rightarrow> rat\<close> and the  inner product is \<open>dot_on UNIV\<close>. Finiteness of $E$ and of the index types of \<open>i\<close>, \<open>j\<close> is not  needed here.\<close> definition polarized :: "('a \<times> 'b) set \<Rightarrow> ('a \<Rightarrow> 'b \<Rightarrow> rat) \<Rightarrow>    ('a \<Rightarrow> 'b \<Rightarrow> 'r::finite \<Rightarrow> rat) \<Rightarrow> bool" where  "polarized E c v \<longleftrightarrow> (\<forall>i j. (i, j) \<notin> E \<longrightarrow> c i j = 0) \<and>     (\<forall>i j k l. dot_on UNIV (v i j) (v k l) + dot_on UNIV (v i l) (v k j) =        2 * (if i = k \<and> j = l then c i j else 0))" lemma dot_on_UNIV_self_zero:  fixes u :: "'r::finite \<Rightarrow> rat"  assumes "dot_on UNIV u u = 0"  shows "u = 0"proof  fix k  have "\<forall>k\<in>UNIV. u k = 0"    using assms dot_on_self_eq_0[of "UNIV :: 'r set" u] by simp  then show "u k = 0 k" by (simp add: zero_fun_def)qed subsection \<open>Support\<close> lemma support:  assumes "polarized E c v" and "(i, j) \<notin> E"  shows "v i j = 0"proof -

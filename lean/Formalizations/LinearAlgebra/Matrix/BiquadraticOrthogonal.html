y) := by  rw [← mulVec_mulVec, ← mulVec_mulVec, dotProduct_mulVec y, vecMul_transpose] /-- Orthogonal substitution in `P_H`: if `Dᵀ D = 1`, then `P_H(x, D y) = P_{Dᵀ H D}(x, y)`. -/theorem P_mulVec_orthogonal [DecidableEq n] {H D : Matrix n n R} (hD : Dᵀ * D = 1)    (x : Fin 2 → R) (y : n → R) :    P H x (D *ᵥ y) = P (Dᵀ * H * D) x y := by  have h1 : (D *ᵥ y) ⬝ᵥ (D *ᵥ y) = y ⬝ᵥ y := by    have := dotProduct_mulVec_congr D 1 y    rwa [one_mulVec, Matrix.mul_one, hD, one_mulVec] at this  simp only [P, h1, dotProduct_mulVec_congr] /-- Transport of sums of squares of bilinear forms under n R)    (h : ∀ x y, P H x y = ∑ i, (x ⬝ᵥ (B i *ᵥ y)) ^ 2) (x : Fin 2 → R) (y : n → R) :    P (Dᵀ * H * D) x y = ∑ i, (x ⬝ᵥ ((B i * D) *ᵥ y)) ^ 2 := by  rw [← P_mulVec_orthogonal hD, h]  simp only [mulVec_mulVec] /-- Signed-graph switching: `D = diagonal s` with `s i * s i = 1` is orthogonal. -/theorem P_switching [DecidableEq n] (H : Matrix n n R) {s : n → R} (hs : ∀ i, s i * s i = 1)    (x : Fin 2 → R) (y : n → R) :    P H x (diagonal s *ᵥ y) = P (diagonal s * H * diagonal s) x y := by  have hD : (diagonal s)ᵀ * diagonal s = 1 := by    rw [diagonal_transpose, diagonal_mul_diagonal, ←
